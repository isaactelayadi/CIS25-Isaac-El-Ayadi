#include <iostream>
using namespace std;

int main() {
    float a, b, c;

    cout << "Enter three values: ";
    cin >> a >> b >> c;

    if (a > b) {
        if (a > c) {
            cout << "The biggest is: " << a << endl;
        }
        else {
            cout << "The biggest is: " << c << endl;
        }
    }
    else {
        if (b > c) {
            cout << "The biggest is: " << b << endl;
        }
        else {
            cout << "The biggest is: " << c << endl;
        }
    }

    return 0;
}
